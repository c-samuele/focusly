\chapter{Introduzione}

\section{Contesto e Motivazione}

Il progetto \textit{Study Planner} è un'applicazione web distribuita progettata per la gestione collaborativa di task di studio, organizzati in gruppi con metriche analitiche e timer di focus pomodoro. L'applicazione è concepita come caso di studio per le problematiche architetturali tipiche delle reti di calcolatori moderne.

\subsection{Problema Affrontato}

Gli studenti necessitano di uno strumento centralizzato per:
\begin{itemize}
    \item Organizzare task di studio per materia/gruppo
    \item Sincronizzare lo stato dei task tra dispositivi
    \item Collaborare in tempo reale con altri studenti
    \item Tracciare metriche di produttività attraverso analytics
    \item Persistere dati in cloud con backup automatico
\end{itemize}

\subsection{Requisiti Funzionali}

\begin{enumerate}
    \item \textbf{Gestione Task}: Creazione, modifica, cancellazione e marcatura di task completati
    \item \textbf{Organizzazione Gruppi}: Raggruppamento task per materia con milestone
    \item \textbf{Sincronizzazione Real-Time}: Aggiornamento istantaneo stato task tra client
    \item \textbf{Persistenza Cloud}: Storage dati su servizio cloud (Firebase, AWS S3, ecc.)
    \item \textbf{Analytics}: Grafici di produttività e metriche di studio
    \item \textbf{Timer Focus}: Timer pomodoro con notifiche real-time
    \item \textbf{Autenticazione}: Sistema di login con JWT token
\end{enumerate}

\subsection{Requisiti Non Funzionali}

\begin{itemize}
    \item \textbf{Scalabilità}: Architettura orizzontale per supportare N utenti
    \item \textbf{Latenza}: Sincronizzazione dati $\leq 200ms$ per WebSocket
    \item \textbf{Disponibilità}: Uptime $\geq 99\%$ con retry automatico
    \item \textbf{Sicurezza}: HTTPS, JWT authentication, input validation
    \item \textbf{Performance}: Caricamento pagina $\leq 2s$ (LCP), bundle size $\leq 100KB$ gzip
\end{itemize}

import axios from 'axios';

const API_BASE_URL = process.env.REACT_APP_API_URL 
  || 'http://localhost:5000/api';

const apiClient = axios.create({
  baseURL: API_BASE_URL,
  timeout: 10000,
  headers: {
    'Content-Type': 'application/json',
  },
});

// Interceptor per aggiungere token JWT
apiClient.interceptors.request.use(config => {
  const token = localStorage.getItem('authToken');
  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }
  return config;
});

// Interceptor per gestire errori
apiClient.interceptors.response.use(
  response => response,
  error => {
    if (error.response?.status === 401) {
      // Token scaduto, redirect login
      window.location.href = '/login';
    }
    return Promise.reject(error);
  }
);

export default apiClient;

\section{Comunicazione Frontend-Backend}

\subsection{Flusso di Dati}

Il flusso di dati segue il pattern unidirezionale:

\begin{enumerate}
    \item \textbf{Evento UI}: Utente interagisce con componente (es. clic su "Crea Task")
    \item \textbf{Azione Store}: Hook chiama azione nel store (es. \texttt{createTask()})
    \item \textbf{API Call}: Servizio HTTP effettua richiesta POST al backend
    \item \textbf{Server Processing}: Backend valida, processa, salva nel DB
    \item \textbf{Response}: Server ritorna dati o errore
    \item \textbf{State Update}: Store aggiorna stato locale
    \item \textbf{Re-render}: React re-renderizza componenti interessati
\end{enumerate}

// controllers/taskController.js
const Task = require('../models/Task');
const redisService = require('../services/redisService');

exports.createTask = async (req, res) => {
  try {
    const { title, description, groupId, dueDate } = req.body;
    const userId = req.user.id; // Da JWT middleware
    
    // Validazione input
    if (!title || !groupId) {
      return res.status(400).json({ 
        error: 'title e groupId obbligatori' 
      });
    }
    
    // Creazione documento
    const task = new Task({
      title,
      description,
      groupId,
      userId,
      dueDate,
      completed: false,
      createdAt: new Date(),
    });
    
    // Salva nel database
    await task.save();
    
    // Invalida cache
    await redisService.del(`tasks:${userId}`);
    
    // Broadcast via WebSocket a tutti i client
    req.io.to(`group:${groupId}`).emit('task:created', task);
    
    res.status(201).json(task);
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
};

exports.getTasks = async (req, res) => {
  try {
    const userId = req.user.id;
    const cacheKey = `tasks:${userId}`;
    
    // Prova cache Redis
    const cached = await redisService.get(cacheKey);
    if (cached) {
      return res.json(JSON.parse(cached));
    }
    
    // Query database
    const tasks = await Task.find({ userId })
      .sort({ createdAt: -1 })
      .lean();
    
    // Salva in cache (TTL 5 minuti)
    await redisService.setex(cacheKey, 300, JSON.stringify(tasks));
    
    res.json(tasks);
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
};

exports.updateTask = async (req, res) => {
  try {
    const { id } = req.params;
    const { title, completed, dueDate } = req.body;
    
    const task = await Task.findByIdAndUpdate(
      id,
      { title, completed, dueDate },
      { new: true, runValidators: true }
    );
    
    // Invalida cache
    await redisService.del(`tasks:${req.user.id}`);
    
    // Broadcast update
    req.io.to(`group:${task.groupId}`).emit('task:updated', task);
    
    res.json(task);
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
};
\end{lstlisting}

\chapter{Comunicazione Real-Time con WebSocket}

\section{WebSocket e Socket.io}

Per la sincronizzazione real-time dei task e del timer focus, utilizziamo \textbf{Socket.io} che fornisce:

\begin{itemize}
    \item Fallback automatico a polling HTTP se WebSocket non disponibile
    \item Reconnection automatica
    \item Rooms per gestire utenti per gruppo
    \item Acknowledgments
\end{itemize}

\section{Implementazione Server}

\begin{lstlisting}
// server.js
const express = require('express');
const { createServer } = require('http');
const { Server: SocketIOServer } = require('socket.io');
const jwt = require('jsonwebtoken');

const app = express();
const httpServer = createServer(app);
const io = new SocketIOServer(httpServer, {
  cors: { origin: 'http://localhost:3000' },
});

// Middleware autenticazione WebSocket
io.use((socket, next) => {
  const token = socket.handshake.auth.token;
  if (!token) return next(new Error('token mancante'));
  
  try {
    const decoded = jwt.verify(token, process.env.JWT_SECRET);
    socket.userId = decoded.id;
    next();
  } catch (error) {
    next(new Error('token non valido'));
  }
});

// WebSocket event handlers
io.on('connection', (socket) => {
  console.log(`Utente ${socket.userId} connesso`);
  
  // Join a gruppo
  socket.on('join:group', (groupId) => {
    socket.join(`group:${groupId}`);
    socket.emit('joined:group', { groupId });
  });
  
  // Timer focus aggiornamento
  socket.on('timer:tick', (data) => {
    io.to(`group:${data.groupId}`).emit('timer:update', {
      taskId: data.taskId,
      timeRemaining: data.timeRemaining,
      userId: socket.userId,
    });
  });
  
  // Task aggiornamento real-time
  socket.on('task:complete', async (taskId) => {
    // Aggiorna in DB
    const task = await Task.findByIdAndUpdate(
      taskId,
      { completed: true, completedAt: new Date() },
      { new: true }
    );
    
    // Notifica altri client
    io.to(`group:${task.groupId}`).emit('task:completed', task);
  });
  
  socket.on('disconnect', () => {
    console.log(`Utente ${socket.userId} disconnesso`);
  });
});

httpServer.listen(5000, () => {
  console.log('Server listening on port 5000');
});
\end{lstlisting}

      reconnectionDelay: 1000,

      reconnectionAttempts: 5,
    });
    
    socket.on('connect', () => {
      console.log('Connesso al server WebSocket');
      if (selectedGroupId) {
        socket.emit('join:group', selectedGroupId);
      }
    });
    
    socket.on('task:updated', (task) => {
      updateTask(task);
    });
    
    socket.on('task:completed', (task) => {
      updateTask({ ...task, completed: true });
    });
    
    socket.on('timer:update', (data) => {
      // Aggiorna stato timer
      useAppStore.setState(state => ({
        timers: {
          ...state.timers,
          [data.taskId]: data.timeRemaining,
        }
      }));
    });
    
    return () => socket.disconnect();
  }, [selectedGroupId]);
};
\end{lstlisting}

\section{Latenza Real-Time}

La latenza end-to-end della sincronizzazione è calcolata come:

$$\Delta t = t_{client\_send} + t_{network} + t_{server\_process} + t_{broadcast}$$

Dove:
\begin{itemize}
    \item $t_{client\_send}$: tempo per serializzare e inviare il messaggio ($\sim 1ms$)
    \item $t_{network}$: latenza di rete TCP/IP ($\sim 50-100ms$ in LAN)
    \item $t_{server\_process}$: elaborazione e query DB ($\sim 10-50ms$)
    \item $t_{broadcast}$: trasmissione ai client ($\sim 20-50ms$)
\end{itemize}

Totale atteso: $\approx 100-200ms$ per aggiornamento sincronizzato tra client.

\section{JWT Token Based Authentication}

L'autenticazione utilizza JSON Web Token (JWT) per statelessness e scalabilità:

\begin{lstlisting}
// middleware/authentication.js
const jwt = require('jsonwebtoken');

exports.authenticateToken = (req, res, next) => {
  const authHeader = req.headers['authorization'];
  const token = authHeader && authHeader.split(' ')[1]; // Bearer <token>
  
  if (!token) {
    return res.status(401).json({ error: 'Token mancante' });
  }
  
  jwt.verify(token, process.env.JWT_SECRET, (err, user) => {
    if (err) {
      return res.status(403).json({ error: 'Token non valido' });
    }
    req.user = user;
    next();
  });
};

// controllers/authController.js
exports.login = async (req, res) => {
  try {
    const { email, password } = req.body;
    
    // Verifica credenziali
    const user = await User.findOne({ email });
    if (!user) {
      return res.status(401).json({ error: 'Credenziali invalide' });
    }
    
    const passwordMatch = await bcrypt.compare(password, user.passwordHash);
    if (!passwordMatch) {
      return res.status(401).json({ error: 'Credenziali invalide' });
    }
    
    // Genera JWT
    const token = jwt.sign(
      { id: user._id, email: user.email },
      process.env.JWT_SECRET,
      { expiresIn: '7d' }
    );
    
    const refreshToken = jwt.sign(
      { id: user._id },
      process.env.JWT_REFRESH_SECRET,
      { expiresIn: '30d' }
    );
    
    res.json({
      token,
      refreshToken,
      user: { id: user._id, email: user.email, name: user.name }
    });
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
};
\end{lstlisting}

\section{Best Practices Sicurezza}

\begin{enumerate}
    \item \textbf{Password Hashing}: bcrypt con salt rounds = 10
    \item \textbf{CORS}: Configurare origins authorized per frontend
    \item \textbf{Rate Limiting}: Max 5 failed login attempts / 15 min
    \item \textbf{Input Validation}: Validazione schema con Joi
    \item \textbf{SQL Injection Prevention}: Prepared statements (Mongoose)
    \item \textbf{XSS Prevention}: Sanitizzazione input, Content Security Policy headers
    \item \textbf{HTTPS/TLS}: Certificati SSL/TLS per produzione (Let's Encrypt)
\end{enumerate}
